// ... (保留之前的 CWA_API_KEY 與 initPWSMonitor 設定) ... function fetchPWSData() { // 呼叫 API 取得 CAP 警報 (此為架構示意) const url = `https://opendata.cwa.gov.tw/api/v1/rest/datastore/W-C0033-001?Authorization=${CWA_API_KEY}`; fetch(url) .then(response => response.json()) .then(data => { const alerts = data.records.record; // 條件過濾：找出影響範圍包含新北市/淡水區的警報 const localAlerts = alerts.filter(alert => { const affectedAreas = alert.hazardConditions.hazard.info.affectedAreas.location; return affectedAreas.some(loc => loc.locationName.includes("新北市") || loc.locationName.includes("淡水")); }); if (localAlerts.length > 0) { const alertData = localAlerts[0]; const alertMessage = alertData.hazardConditions.hazard.info.event; // 【核心修改】判斷警報的 Status 欄位 (CAP 標準) // 假設 API 回傳的狀態欄位名為 'status' const alertStatus = alertData.status || "Actual"; // 預設當作真實事件 if (alertStatus === "Exercise" || alertStatus === "Test") { // 若為演習或測試，觸發【演習專屬 UI】 triggerExerciseUI(alertMessage); } else if (alertStatus === "Actual") { // 若為真實事件，觸發【真實警報 UI】 triggerRealPWSUI(alertMessage); } } }) .catch(err => { console.log("PWS 監控待命中 (API 尚未填入或無最新警報)"); }); } // ========================================== // 1. 【演習/測試模式】黃色溫和提示，不響刺耳警報 // ========================================== function triggerExerciseUI(alertMessage) { const overlay = document.getElementById('pws-overlay'); const msgBox = document.getElementById('pws-msg'); // 改變顏色為黃色/橘色，降低恐慌感 overlay.style.backgroundColor = "rgba(255, 165, 0, 0.4)"; overlay.style.borderColor = "#ffa500"; overlay.style.animation = "exercise-flash 2s infinite alternate"; // 閃爍變慢 msgBox.innerHTML = `📢 國家級警報 (演習/測試)

${alertMessage}
此為演習測試，請民眾勿驚慌。`; overlay.style.display = 'flex'; // 跑馬燈更新為橘色演習模式 const tickerBox = document.getElementById('ticker-box'); document.getElementById('ticker-text').innerText = `【演習測試】${alertMessage}，此為演習訊息，請民眾保持鎮定。`; tickerBox.style.backgroundColor = "rgba(255, 165, 0, 0.15)"; tickerBox.style.borderColor = "#ffa500"; setTimeout(() => { overlay.style.display = 'none'; }, 15000); } // ========================================== // 2. 【真實災害模式】紅色刺耳警報 (保留原本設計) // ========================================== function triggerRealPWSUI(alertMessage) { const overlay = document.getElementById('pws-overlay'); const msgBox = document.getElementById('pws-msg'); const siren = document.getElementById('pws-siren'); // 恢復紅色警戒狀態 overlay.style.backgroundColor = "rgba(255, 0, 0, 0.4)"; overlay.style.borderColor = "red"; overlay.style.animation = "pws-flash 1s infinite alternate"; msgBox.innerHTML = `⚠️ 國家級警報 (真實事件) ⚠️

${alertMessage}
請立即啟動緊急應變程序！`; overlay.style.display = 'flex'; siren.play().catch(e => console.log("音效需點擊畫面後生效")); document.getElementById('ticker-text').innerText = `【國家級警報】${alertMessage}，請立即啟動應變機制！`; document.getElementById('ticker-box').classList.add('emergency'); setTimeout(() => { overlay.style.display = 'none'; siren.pause(); siren.currentTime = 0; }, 15000); }
